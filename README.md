  for(let d=0;d<8;d++){
    const day=new Date(now); day.setDate(now.getDate()+d);
    if(!RUN_DOW.includes(day.getDay())) continue;
    const t=new Date(day); t.setHours(h,m,0,0);
    if(t>now) return t;
  }
}
function fmtCD(ms){
  let s=Math.max(0,Math.round(ms/1000));
  const days=Math.floor(s/86400); s%=86400;
  const hh=Math.floor(s/3600), mm2=Math.floor(s%3600/60), ss=s%60;
  return (days>0?days+' ngày ':'')+pad(hh)+':'+pad(mm2)+':'+pad(ss);
}
function fmtDate(t){ return DOW[t.getDay()]+', ngày '+pad(t.getDate())+' tháng '+pad(t.getMonth()+1); }
function tickBoard(){
  const tS=nextOccur(8,0), tC=nextOccur(13,30);
  if(tS){ $('#dSang').textContent=fmtDate(tS); $('#cdSang').textContent=fmtCD(tS-new Date()); }
  if(tC){ $('#dChieu').textContent=fmtDate(tC); $('#cdChieu').textContent=fmtCD(tC-new Date()); }
}
tickBoard(); setInterval(tickBoard,1000);

(function(){
  const today=new Date(); today.setHours(0,0,0,0);
  let cur=new Date(today.getFullYear(),today.getMonth(),1);
  const FULL_DAYS=[8,15,22];
  const daysEl=$('#calDays'), title=$('#calTitle');
  function render(){
    const y=cur.getFullYear(), m=cur.getMonth();
    title.textContent='Tháng '+(m+1)+' / '+y;
    daysEl.innerHTML='';
    const start=(new Date(y,m,1).getDay()+6)%7;
    for(let i=0;i<start;i++) daysEl.insertAdjacentHTML('beforeend','<span class="blank"></span>');
    const last=new Date(y,m+1,0).getDate();
    for(let d=1;d<=last;d++){
      const dt=new Date(y,m,d), dow=dt.getDay(), isPast=dt<today;
      const open=RUN_DOW.includes(dow);
      const full=open && m===today.getMonth() && y===today.getFullYear() && FULL_DAYS.includes(d);
      const b=document.createElement('button');
      b.innerHTML=d+(open&&!full?'<small>2 ca</small>':'');
      b.className=(isPast?'is-past ':full?'is-full ':open?'is-open ':'')+(dt.getTime()===today.getTime()?'is-today ':'');
      if(open&&!full&&!isPast) b.addEventListener('click',()=>select(b,dt));
      daysEl.appendChild(b);
    }
  }
  function select(b,dt){
    $$('.cal-days .sel').forEach(x=>x.classList.remove('sel')); b.classList.add('sel');
    $('#cpDate').textContent=DOW[dt.getDay()]+', '+pad(dt.getDate())+'/'+pad(dt.getMonth()+1)+'/'+dt.getFullYear();
    $('#cpNote').textContent='Ngày có 2 ca khởi hành. Vé còn — nên đặt trước để chọn chỗ đẹp.';
    $('#cpShifts').innerHTML=
      '<div class="shift"><span>🌅 Ca sáng <b>08:00 – 11:30</b></span><span class="tag-ok">✓ Còn vé</span></div>'+
      '<div class="shift"><span>🌇 Ca chiều <b>13:30 – 17:00</b></span><span class="tag-ok">✓ Còn vé</span></div>';
  }
  $('#calPrev').onclick=()=>{cur=new Date(cur.getFullYear(),cur.getMonth()-1,1);render()};
  $('#calNext').onclick=()=>{cur=new Date(cur.getFullYear(),cur.getMonth()+1,1);render()};
  render();
})();

(function(){
  const lb=$('#lb'), img=$('#lbImg'), cap=$('#lbCap');
  let groups={}, cur=[], idx=0;
  $$('[data-lb]').forEach(el=>{ (groups[el.dataset.lb]=groups[el.dataset.lb]||[]).push(el);
    el.addEventListener('click',()=>open(el.dataset.lb,groups[el.dataset.lb].indexOf(el))); });
  function open(g,i){ cur=groups[g]; idx=i; show(); lb.classList.add('open'); document.body.style.overflow='hidden'; }
  function show(){ const el=cur[idx], im=el.querySelector('img')||el;
    img.src=(im.src||im.currentSrc).replace(/\/\d+\/\d+$/,'/1200/800'); img.alt=el.dataset.cap||'';
    cap.textContent=(el.dataset.cap||'')+'  ·  '+(idx+1)+'/'+cur.length; }
  function close(){ lb.classList.remove('open'); document.body.style.overflow=''; }
  $('.lb-x').onclick=close; $('.lb-p').onclick=()=>{idx=(idx-1+cur.length)%cur.length;show()}; $('.lb-n').onclick=()=>{idx=(idx+1)%cur.length;show()};
  lb.addEventListener('click',e=>{if(e.target===lb)close()});
  addEventListener('keydown',e=>{ if(!lb.classList.contains('open'))return;
    if(e.key==='Escape')close(); if(e.key==='ArrowLeft')$('.lb-p').click(); if(e.key==='ArrowRight')$('.lb-n').click(); });
})();
